void WriteLog(string type, string message, double price=0, double lots=0) {
   string time = TimeToString(TimeCurrent(), TIME_DATE|TIME_SECONDS);
   string balance = DoubleToString(AccountInfoDouble(ACCOUNT_BALANCE), 2);
   string logText = StringFormat("[%s] %s | Price: %.2f | Vol: %.2f | Bal: %s", type, message, price, lots, balance);
   Print(logText); 
   if(InpEnableFileLog && fileHandle != INVALID_HANDLE) {
      FileWrite(fileHandle, time, type, message, DoubleToString(price, 2), DoubleToString(lots, 2), balance);
      FileFlush(fileHandle); 
   }
}

double GetDynamicMoneyValue(double percent) {
   double balance = AccountInfoDouble(ACCOUNT_BALANCE);
   return (balance * percent) / 100.0;
}

double GetEffectiveATR() {
   double rawATR = bufferATR[0];
   double minATR = SymbolInfoDouble(_Symbol, SYMBOL_POINT) * 150; 
   return MathMax(rawATR, minATR);
}

bool CheckFreeMargin(double lots, long type) {
   double marginRequired = 0.0;
   if(!OrderCalcMargin((ENUM_ORDER_TYPE)type, _Symbol, lots, SymbolInfoDouble(_Symbol, SYMBOL_ASK), marginRequired)) return false;
   if(AccountInfoDouble(ACCOUNT_MARGIN_FREE) < marginRequired * 1.1) return false;
   return true;
}

double CalculateLots(bool isRecoveryPhase) {
   double equity = AccountInfoDouble(ACCOUNT_EQUITY);
   double ratio = equity / InpBaseRisk;
   double lots = ratio * InpBaseLots;
   
   double volStep = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_STEP);
   lots = MathFloor(lots / volStep) * volStep;
   double minLot = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MIN);
   double maxLot = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MAX);
   
   if(lots < minLot) lots = minLot;
   if(lots > maxLot) lots = maxLot;
   
   double currentCap = isRecoveryPhase ? InpMaxRecoveryLots : InpMaxEntryLots;
   if(lots > currentCap) lots = currentCap;
   
   return lots;
}

int GetH4Structure() {
   if(!InpUseH4_Struct) return 0; 
   double h1=0, h2=0, l1=0, l2=0;
   int c=0;
   for(int i=2; i<100; i++) {
      double f = iFractalsGet(handleFractalsH4, 0, i);
      if(f>0) { if(c==0) h1=f; else { h2=f; break; } c++; }
   }
   c=0;
   for(int i=2; i<100; i++) {
      double f = iFractalsGet(handleFractalsH4, 1, i);
      if(f>0) { if(c==0) l1=f; else { l2=f; break; } c++; }
   }
   if(h1>h2 && l1>l2) return 1;  
   if(l1<l2 && h1<h2) return -1; 
   if(l1<l2) return -1; 
   if(h1>h2) return 1;
   return 0;
}

int CheckM15Breakout() {

   for(int i=3; i<30; i++) {
      double u = iFractalsGet(handleFractals, 0, i);

      if(upF==0 && u>0) upF=u;
      if(loF==0 && l>0) loF=l;
      if(upF>0 && loF>0) break;
   }
   double c1 = iClose(_Symbol, _Period, 1);
   if(upF>0 && c1>upF) return 1;
   if(loF>0 && c1<loF) return -1;
   return 0;
}

double CalculateCurrentBasketNetProfit() {
   double netPL = 0;
   int total = PositionsTotal();
   for(int i=0; i<total; i++) {
      ulong t = PositionGetTicket(i);
      if(PositionSelectByTicket(t) && PositionGetInteger(POSITION_MAGIC) == magicNum) {
         netPL += PositionGetDouble(POSITION_PROFIT);
         netPL += PositionGetDouble(POSITION_SWAP);
         double vol = PositionGetDouble(POSITION_VOLUME);
         double commissionCost = vol * InpCommissionPerLot;
         netPL -= commissionCost; 
      }
   }
   return netPL;
}

double CalculateVolatilityRiskInMoney() {
    double totalLots = 0;
    int total = PositionsTotal();
    for(int i=0; i<total; i++) {
        ulong t = PositionGetTicket(i);
        if(PositionSelectByTicket(t) && PositionGetInteger(POSITION_MAGIC) == magicNum) totalLots += PositionGetDouble(POSITION_VOLUME);
    }
    if(totalLots <= 0) return 1.0;
    double atrPoints = GetEffectiveATR() / _Point;
    double tickValue = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_VALUE);
    if(tickValue == 0) tickValue = 1.0; 
    return atrPoints * totalLots * tickValue;
}

void CheckOp3Healing() {
   if(!InpUseHealer) return;
   
   datetime startLookback = TimeCurrent() - 3600; 
   if(HistorySelect(startLookback, TimeCurrent())) {
      int deals = HistoryDealsTotal();
      for(int i=deals-1; i>=0; i--) {
         ulong ticket = HistoryDealGetTicket(i);
         if(ticket <= lastHealedTicket) break; 
         
         if(HistoryDealGetInteger(ticket, DEAL_MAGIC) == magicNum) {
            if(HistoryDealGetInteger(ticket, DEAL_ENTRY) == DEAL_ENTRY_OUT) {
               double profit = HistoryDealGetDouble(ticket, DEAL_PROFIT);
               if(profit > 0) {
                  ApplyHealing(profit);
                  lastHealedTicket = ticket; 
                  return; 
               }
            }
         }
      }
   }
}

bool ApplyHealing(double profitAvailable) {
   ulong worstTicket = 0;
   double worstProfit = 1000000;
   double worstPriceOpen = 0;
   long worstType = 0;
   
   for(int i=PositionsTotal()-1; i>=0; i--) {
      ulong t = PositionGetTicket(i);
      if(PositionSelectByTicket(t) && PositionGetInteger(POSITION_MAGIC) == magicNum) {
         double p = PositionGetDouble(POSITION_PROFIT);
         if(p < worstProfit) {
            worstProfit = p;
            worstTicket = t;
            worstPriceOpen = PositionGetDouble(POSITION_PRICE_OPEN);
            worstType = PositionGetInteger(POSITION_TYPE);
         }
      }
   }
   
   if(worstTicket == 0 || worstProfit >= 0) return true; 

   double currentPrice = (worstType == POSITION_TYPE_BUY) ? SymbolInfoDouble(_Symbol, SYMBOL_BID) : SymbolInfoDouble(_Symbol, SYMBOL_ASK);
   double diffPoints = MathAbs(currentPrice - worstPriceOpen) / _Point;
   if(diffPoints == 0) return true;
   
   double tickValue = SymbolInfoDouble(_Symbol, SYMBOL_TRADE_TICK_VALUE);
   if(tickValue == 0) tickValue = 1.0;
   
   double minLot = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MIN);
   double costOfMinLot = (diffPoints * tickValue) * minLot; 

   if(costOfMinLot > profitAvailable) return true; 

   double budget = profitAvailable * 0.90; 
   double lotsToClose = budget / (diffPoints * tickValue);
   
   double volStep = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_STEP);
   lotsToClose = MathFloor(lotsToClose / volStep) * volStep;
   
   if(lotsToClose >= minLot) {
      if(trade.PositionClosePartial(worstTicket, lotsToClose)) {
         WriteLog("HEALER", "Amputacion Tactica. Lotes: " + DoubleToString(lotsToClose, 2), worstProfit, budget);
      }
   }
   return true;
}

void RunGrinder() {
    if(!InpUseGrinder) return;
    if(!CheckFreeMargin(InpGrinderLots, ORDER_TYPE_BUY)) return;

    // A. Gestión de Time Stop (Limpieza)
    int grinderOps = 0;
    for(int i=PositionsTotal()-1; i>=0; i--) {
        ulong t = PositionGetTicket(i);
        if(PositionSelectByTicket(t) && PositionGetInteger(POSITION_MAGIC) == magicNum) {
            // IDENTIFICAR GRINDER POR LOTAJE
            if(PositionGetDouble(POSITION_VOLUME) == InpGrinderLots) {
                grinderOps++;
                // Time Stop Logic
                datetime openTime = (datetime)PositionGetInteger(POSITION_TIME);
                if(TimeCurrent() - openTime > (InpGrinderTimeStop * 60)) {
                    double p = PositionGetDouble(POSITION_PROFIT);
                    if(p < 0) { 
                        trade.PositionClose(t);
                        WriteLog("GRINDER", "TimeStop Activado. Limpiando zona.", p, 0);
                        return; 
                    }
                }
            }
        }
    }
    if(grinderOps >= 1) return; // Solo 1 Grinder a la vez

    // B. Lógica de Entrada (SmartCut M5)
    double adx = bufferG_ADX[0];
    double rsi = bufferG_RSI[0];
    double closeM5 = iClose(_Symbol, PERIOD_M5, 0);
    double maM5 = bufferG_MA[0]; 

    // B1. Modo Scalper (Reversión) - ADX Bajo
    if(adx < InpGrinderADX_Trend) {
        if(rsi < InpGrinderRSI_OS) { 
             trade.Buy(InpGrinderLots, _Symbol, SymbolInfoDouble(_Symbol, SYMBOL_ASK), 0, 0, "Grinder Scalp Buy");
        }
        else if(rsi > InpGrinderRSI_OB) { 
             trade.Sell(InpGrinderLots, _Symbol, SymbolInfoDouble(_Symbol, SYMBOL_BID), 0, 0, "Grinder Scalp Sell");
        }
    }
    // B2. Modo Surfer (Tendencia) - ADX Alto
    else {
        if(closeM5 > maM5 && rsi < 70) { 
             trade.Buy(InpGrinderLots, _Symbol, SymbolInfoDouble(_Symbol, SYMBOL_ASK), 0, 0, "Grinder Surf Buy");
        }
        else if(closeM5 < maM5 && rsi > 30) { 
             trade.Sell(InpGrinderLots, _Symbol, SymbolInfoDouble(_Symbol, SYMBOL_BID), 0, 0, "Grinder Surf Sell");
        }
    }
}

// --- NUEVA FUNCION: TRAILING EXCLUSIVO PARA GRINDER ---
void ManageGrinderTrailing() {
    if(!InpGrinderUseTrail) return;

    for(int i=PositionsTotal()-1; i>=0; i--) {
        ulong t = PositionGetTicket(i);
        if(PositionSelectByTicket(t) && PositionGetInteger(POSITION_MAGIC) == magicNum) {
            // SOLO APLICAMOS AL GRINDER (Identificado por lotaje)
            if(PositionGetDouble(POSITION_VOLUME) == InpGrinderLots) {
                
                long type = PositionGetInteger(POSITION_TYPE);
                double currentPrice = PositionGetDouble(POSITION_PRICE_CURRENT);
                double openPrice = PositionGetDouble(POSITION_PRICE_OPEN);
                double sl = PositionGetDouble(POSITION_SL);
                
                double points = 0;
                if(type == POSITION_TYPE_BUY) points = (currentPrice - openPrice) / _Point;
                else points = (openPrice - currentPrice) / _Point;

                // Lógica 2-SPEED
                int activeDist = InpGrinderTrailDist; // Velocidad Normal
                if(points >= InpGrinderTurboTrig) activeDist = InpGrinderTurboDist; // Velocidad Turbo

                if(points >= InpGrinderTrailStart) {
                    double newSL = 0;
                    bool update = false;

                    if(type == POSITION_TYPE_BUY) {
                        newSL = NormalizeDouble(currentPrice - (activeDist * _Point), _Digits);
                        if(newSL > openPrice && (sl == 0 || newSL > sl)) update = true;
                    }
                    else {
                        newSL = NormalizeDouble(currentPrice + (activeDist * _Point), _Digits);
                        if(newSL < openPrice && (sl == 0 || newSL < sl)) update = true;
                    }

                    if(update) trade.PositionModify(t, newSL, 0);
                }
            }
        }
    }
}

void ManagePartialProfitBanking() {
    if(!InpUseUnwindMode) return;
    int total = PositionsTotal();
    ulong bestTicket = 0;
    double bestDist = -1.0; 
    double bestMoney = -999999;
    int myCount = 0;
    double currentATR = GetEffectiveATR();
    for(int i=0; i<total; i++) {
       ulong t = PositionGetTicket(i);
       if(PositionSelectByTicket(t) && PositionGetInteger(POSITION_MAGIC) == magicNum) {
          myCount++;
          double profit = PositionGetDouble(POSITION_PROFIT) + PositionGetDouble(POSITION_SWAP);
          if(profit <= 0) continue;
          double netProfit = profit - (PositionGetDouble(POSITION_VOLUME) * InpCommissionPerLot);
          if(netProfit > bestMoney) bestMoney = netProfit;
          long type = PositionGetInteger(POSITION_TYPE);
          double openPrice = PositionGetDouble(POSITION_PRICE_OPEN);
          double currentPrice = PositionGetDouble(POSITION_PRICE_CURRENT);
          double rawDist = MathAbs(currentPrice - openPrice);
          double atrMultiples = rawDist / currentATR;
          if(atrMultiples > bestDist) { bestDist = atrMultiples; bestTicket = t; }
       }
    }
    if(myCount < 2) return;
    if(bestTicket > 0) {
        if(bestDist >= InpUnwindATR_Mult || bestMoney >= InpUnwindMoney_Floor) {
            trade.PositionClose(bestTicket);
            WriteLog("UNWIND", "Profit Banking. Money: " + DoubleToString(bestMoney, 2), 0, 0);
        }
    }
}

    }

    double currentATR = GetEffectiveATR();
    double minDistance = currentATR * activeATRMult; 
    double rsi = bufferRSI[0];
    string comment = "Sentinel Op 4 (L1)";
    if(absDD > level2_Money) comment = "Sentinel Op 4 (L2)";
    if(absDD > level3_Money) comment = "Sentinel Op 4 (L3-CRITICO)";

    bool signal = false;
    
    // NOTA: Op4 es RESCATE, por tanto DEBE seguir la dirección de Op3 para promediar.
    if(typeOp3 == POSITION_TYPE_SELL) { 
        double dist = SymbolInfoDouble(_Symbol, SYMBOL_BID) - priceOp3;
        if(dist >= minDistance) {
            if(ignoreRSI || rsi > InpRescueRSI) {
               if(!forceEntry || IsM5MomentumFavorable(POSITION_TYPE_SELL)) signal = true;
            }
        }
        if(signal && CheckFreeMargin(volOp3, ORDER_TYPE_SELL)) {
            trade.Sell(volOp3, _Symbol, SymbolInfoDouble(_Symbol, SYMBOL_BID), 0, 0, comment);
        }
    }
    else if(typeOp3 == POSITION_TYPE_BUY) { 
        double dist = priceOp3 - SymbolInfoDouble(_Symbol, SYMBOL_ASK);
        if(dist >= minDistance) {
            if(ignoreRSI || rsi < (100 - InpRescueRSI)) {
               if(!forceEntry || IsM5MomentumFavorable(POSITION_TYPE_BUY)) signal = true;
            }
        }
        if(signal && CheckFreeMargin(volOp3, ORDER_TYPE_BUY)) {
            trade.Buy(volOp3, _Symbol, SymbolInfoDouble(_Symbol, SYMBOL_ASK), 0, 0, comment);
        }
    }
}

// ==================================================================
// ON TICK (CORE)
// ==================================================================
void OnTick()
{
   if(CopyBuffer(handleATR, 0, 0, 3, bufferATR) < 0) return;
   if(CopyBuffer(handleMA, 0, 0, 3, bufferMA) < 0) return;
   if(CopyBuffer(handleRSI, 0, 0, 3, bufferRSI) < 0) return;
   if(CopyBuffer(handleGrinderMA, 0, 0, 2, bufferG_MA) < 0) return;
   if(CopyBuffer(handleGrinderADX, 0, 0, 2, bufferG_ADX) < 0) return;
   if(CopyBuffer(handleGrinderRSI, 0, 0, 2, bufferG_RSI) < 0) return;

   if(!CheckSpread()) return;

   CheckOp3Healing();
   ManagePartialProfitBanking(); 
   ManageGrinderTrailing(); // NUEVO: Gestión de Trailing para Grinder

   int totalPositions = PositionsTotal();
   int buyCount = 0, sellCount = 0;
   
   for(int i = totalPositions - 1; i >= 0; i--) {
      ulong ticket = PositionGetTicket(i);
      if(PositionSelectByTicket(ticket) && PositionGetInteger(POSITION_MAGIC) == magicNum) {
         if(PositionGetInteger(POSITION_TYPE) == POSITION_TYPE_BUY) buyCount++;
         if(PositionGetInteger(POSITION_TYPE) == POSITION_TYPE_SELL) sellCount++;
      }
   }
   int myPositions = buyCount + sellCount;

   double currentATR = GetEffectiveATR();

   // MONITOR DE SALIDA
   if(myPositions > 0) {
      double currentNetPL = CalculateCurrentBasketNetProfit();
      double dynamicTarget = GetDynamicMoneyValue(InpBasketPercent);
      if(dynamicTarget < 2.0) dynamicTarget = 2.0; 
      
      if(currentNetPL >= dynamicTarget) {
         if(currentNetPL > maxCyclePeak) maxCyclePeak = currentNetPL;
         
         double allowedRetrace = InpFixedRetrace;
         if(InpUseDynamicRetrace) {
             double volMoney = CalculateVolatilityRiskInMoney();
             allowedRetrace = volMoney * InpRetraceATRMult; 
             if(allowedRetrace < 1.0) allowedRetrace = 1.0;
             if(allowedRetrace > 50.0) allowedRetrace = 50.0;
         }
         
         if(maxCyclePeak - currentNetPL >= allowedRetrace) {
             CloseAllPositions("Basket Profit Trail");
             return;
         }
      }
      
      if(myPositions >= 4 && InpUseRescueMode) {
          double rescueTarget = GetDynamicMoneyValue(InpRescueTarget_Pct);
          if(currentNetPL >= rescueTarget) {
              CloseAllPositions("Rescue Mission Success");
              return;
          }
      }
   } else {
       maxCyclePeak = 0.0; 
   }

   MqlDateTime dt; TimeCurrent(dt);
   bool isFridayMode = (InpCloseFriday && dt.day_of_week == 5 && dt.hour >= InpFridayHour);
   if(isFridayMode && myPositions > 0 && CalculateCurrentBasketNetProfit() > 0) { 
       CloseAllPositions("Viernes Close"); return; 
   }
   bool isWeeklyStartWait = (dt.day_of_week == 0 || (dt.day_of_week == 1 && dt.hour < InpMondayStartHour));

   int structH4 = GetH4Structure();
   int signalM15 = CheckM15Breakout();
   
   // --- ESTADO 0: ENTRY ---
   if(myPositions == 0) {
      if(isFridayMode || isWeeklyStartWait) return;
      if(lastRecoveryCloseTime > 0 && (TimeCurrent() - lastRecoveryCloseTime < InpCooldownSeconds)) return;
      
      double lots = CalculateLots(false);
      
      if(structH4 >= 0 && signalM15 == 1 && IsPriceGoodEntry(POSITION_TYPE_BUY)) {
         if(bufferRSI[0] < InpEntryRSI_Max) {
             if(CheckFreeMargin(lots, ORDER_TYPE_BUY)) {
                 trade.Buy(lots, _Symbol, SymbolInfoDouble(_Symbol, SYMBOL_ASK), 0, 0, "SMC Buy Entry");
             }
         }
      }
      else if(structH4 <= 0 && signalM15 == -1 && IsPriceGoodEntry(POSITION_TYPE_SELL)) {
         if(bufferRSI[0] > InpEntryRSI_Min) {
             if(CheckFreeMargin(lots, ORDER_TYPE_SELL)) {
                 trade.Sell(lots, _Symbol, SymbolInfoDouble(_Symbol, SYMBOL_BID), 0, 0, "SMC Sell Entry");
             }
         }
      }
   }

   // --- ESTADO 1: HEDGE MONITOR ---
   else if(myPositions == 1) {
      for(int i = totalPositions - 1; i >= 0; i--) {
         ulong ticket = PositionGetTicket(i);
         if(PositionSelectByTicket(ticket) && PositionGetInteger(POSITION_MAGIC) == magicNum) {
            double openPrice = PositionGetDouble(POSITION_PRICE_OPEN);
            long type = PositionGetInteger(POSITION_TYPE);
            double currentLots = PositionGetDouble(POSITION_VOLUME);
            double currentPrice = PositionGetDouble(POSITION_PRICE_CURRENT);
            double sl = PositionGetDouble(POSITION_SL);
            
            double profitPts = (type==POSITION_TYPE_BUY) ? (currentPrice-openPrice)/_Point : (openPrice-currentPrice)/_Point;
            double lossPts   = (type==POSITION_TYPE_BUY) ? (openPrice-currentPrice)/_Point : (currentPrice-openPrice)/_Point;
            
            if(InpUseSmartTrail && profitPts > (currentATR*InpTrailActivate)/_Point) {
                double trailDist = (currentATR*InpTrailDistATR);
                double newSL = (type==POSITION_TYPE_BUY) ? currentPrice - trailDist : currentPrice + trailDist;
                newSL = NormalizeDouble(newSL, _Digits);
                bool modify = false;
                if(type==POSITION_TYPE_BUY && newSL > openPrice && (sl==0 || newSL > sl)) modify=true;
                if(type==POSITION_TYPE_SELL && newSL < openPrice && (sl==0 || newSL < sl)) modify=true;
                if(modify) trade.PositionModify(ticket, newSL, 0);
            }
            
            double atrPoints = currentATR / _Point;
            double dynamicDist = atrPoints * InpHedgeATRMult;
            double activeHedgeDist = MathMax((double)InpHedgeDist, dynamicDist);
            
            if(lossPts >= activeHedgeDist) {
               trade.PositionModify(ticket, 0, 0); 
               if(type == POSITION_TYPE_BUY) {
                   if(CheckFreeMargin(currentLots, ORDER_TYPE_SELL)) trade.Sell(currentLots, _Symbol, SymbolInfoDouble(_Symbol, SYMBOL_BID), 0, 0, "Hedge Lock");
               }
               else {
                   if(CheckFreeMargin(currentLots, ORDER_TYPE_BUY)) trade.Buy(currentLots, _Symbol, SymbolInfoDouble(_Symbol, SYMBOL_ASK), 0, 0, "Hedge Lock");
               }
            }
            break; 
         }
      }
   }

   // --- ESTADO 2: RECOVERY ---
   else if(myPositions == 2) {
      datetime op2Time = 0;
      for(int i=0; i<totalPositions; i++) {
         ulong t = PositionGetTicket(i);
         if(PositionSelectByTicket(t) && PositionGetInteger(POSITION_MAGIC) == magicNum) {
             datetime pt = (datetime)PositionGetInteger(POSITION_TIME);
             if(pt > op2Time) op2Time = pt;
         }
      }
      if(TimeCurrent() - op2Time < InpMinTechWait) return;

      double baseLots = CalculateLots(false);
      double recoveryLots = baseLots * 2.0; 
      if(recoveryLots > InpMaxRecoveryLots) recoveryLots = InpMaxRecoveryLots;

      if(bufferATR[0] > bufferATR[1]) {
         bool m5Buy  = IsM5MomentumFavorable(POSITION_TYPE_BUY);
         bool m5Sell = IsM5MomentumFavorable(POSITION_TYPE_SELL);
         
         if(signalM15 == 1 || (m5Buy && signalM15 != -1)) { 
             if(CheckFreeMargin(recoveryLots, ORDER_TYPE_BUY)) trade.Buy(recoveryLots, _Symbol, SymbolInfoDouble(_Symbol, SYMBOL_ASK), 0, 0, "Recovery Buy (V-Shape)");
         }
         
         if(signalM15 == -1 || (m5Sell && signalM15 != 1)) { 
             if(CheckFreeMargin(recoveryLots, ORDER_TYPE_SELL)) trade.Sell(recoveryLots, _Symbol, SymbolInfoDouble(_Symbol, SYMBOL_BID), 0, 0, "Recovery Sell (V-Shape)");
         }
      }
   }
   
   // --- ESTADO 3+: TRAILING FLUIDO + SENTINEL + BIO-REACTOR ---
   else if(myPositions >= 3) {
      
      // 1. Sentinel Op4
      if(myPositions == 3) CheckRescueEntry();
      
      // 2. Grinder Bio-Reactor (SmartCut Inside)
      if(myPositions >= 4) RunGrinder();

      // 3. Trailing Op3 (Turbo)
      if(InpUseOp3Trail) {
         ulong ticketOp3 = 0;
         datetime lastT = 0;
         for(int i=0; i<totalPositions; i++) {
            ulong t = PositionGetTicket(i);
            if(PositionSelectByTicket(t) && PositionGetInteger(POSITION_MAGIC) == magicNum) {
                // Filtramos que NO sea un trade del Grinder
                if(PositionGetDouble(POSITION_VOLUME) != InpGrinderLots) {
                    if(PositionGetInteger(POSITION_TIME) > lastT) {
                        lastT = (datetime)PositionGetInteger(POSITION_TIME);
                        ticketOp3 = t;
                    }
                }
            }
         }
         
         if(ticketOp3 > 0) {
             double currentProfit = PositionGetDouble(POSITION_PROFIT) + PositionGetDouble(POSITION_SWAP);
             if(currentProfit > 0) {
                 long type = PositionGetInteger(POSITION_TYPE);
                 double curPrice = PositionGetDouble(POSITION_PRICE_CURRENT);
                 double openPrice = PositionGetDouble(POSITION_PRICE_OPEN);
                 double sl = PositionGetDouble(POSITION_SL);
                 double pts = (type==POSITION_TYPE_BUY) ? (curPrice-openPrice)/_Point : (openPrice-curPrice)/_Point;
                 
                 double startPts = (currentATR * InpOp3_StartATR) / _Point;
                 double activeDist = (currentATR * InpOp3_BaseDistATR) / _Point;
                 double turboTriggerPts = (currentATR * InpOp3_TurboTriggerATR) / _Point;
                 
                 if(pts >= turboTriggerPts) activeDist = (currentATR * InpOp3_TurboDistATR) / _Point;
                 if(pts > startPts) {
                     double newSL = (type==POSITION_TYPE_BUY) ? curPrice - (activeDist*_Point) : curPrice + (activeDist*_Point);
                     newSL = NormalizeDouble(newSL, _Digits);
                     bool mod = false;
                     if(type==POSITION_TYPE_BUY && (sl==0 || newSL > sl)) mod=true;
                     if(type==POSITION_TYPE_SELL && (sl==0 || newSL < sl)) mod=true;
                     if(mod) trade.PositionModify(ticketOp3, newSL, 0);
                 }
             }
         }
      }
   }
}
